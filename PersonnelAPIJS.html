//---------------------------------- // Personnel Apps Script API //---------------------------------- const PersonnelAPI = { //---------------------------------- // Load Personnel //---------------------------------- loadPersonnel() { return new Promise((resolve, reject) => { google.script.run .withSuccessHandler(response => { try { resolve(useSheetFullNamePersonnel_(response)); } catch (error) { reject(normalizeServerError_(error)); } }) .withFailureHandler(error => { reject(normalizeServerError_(error)); }) .getPersonnelTableV2Data(); }); }, //---------------------------------- // Apply Visible Rows //---------------------------------- applyVisibleRows(payload) { return new Promise((resolve, reject) => { google.script.run .withSuccessHandler(resolve) .withFailureHandler(error => { reject(normalizeServerError_(error)); }) .applyPersonnelVisibleRows(payload); }); }, //---------------------------------- // Clear Personnel Cache //---------------------------------- clearCache() { return Promise.resolve({ success: true, skipped: true, message: "The main personnel loader now reads directly from Google Sheets.", }); }, }; //---------------------------------- // Keep the editable FULL NAME as the app display value //---------------------------------- // Canonical columns remain in each record for generator compatibility, but // they do not override changes made to FULL NAME in the LIST sheet. function useSheetFullNamePersonnel_(response) { const source = response && typeof response === "object" ? response : {}; const records = Array.isArray(source.records) ? source.records : []; const sheetRecords = records.map(record => { const rawFullName = String( record?.["Raw Full Name"] || record?.["Full Name"] || "" ).trim(); const canonicalName = String( record?.["CANONICAL NAME"] || record?.canonicalName || "" ).trim(); const canonicalNameWithRank = String( record?.["CANONICAL NAME WITH RANK"] || record?.canonicalNameWithRank || "" ).trim(); const displayName = rawFullName || canonicalNameWithRank; const personnelSearchKey = displayName .replace(/[^A-Za-z0-9]/g, "") .toUpperCase(); return { ...record, "Raw Full Name": rawFullName, "Full Name": displayName, displayName, canonicalName, canonicalNameWithRank, personnelSearchKey, searchKey: personnelSearchKey, personnelIdentityKey: personnelSearchKey, }; }); return { ...source, records: sheetRecords, nameNormalization: { engine: "bypassed", source: "editable-full-name", normalizedCount: sheetRecords.length, canonicalColumnsAuthoritative: false, }, }; } //---------------------------------- // Normalize Server Error //---------------------------------- function normalizeServerError_(error) { if (error instanceof Error) return error; return new Error( error?.message || String(error) || "Unknown Apps Script error." ); }
